  23  def digits10(s):
  24      d = "".join(ch for ch in s if ch.isdigit())
  25      if len(d) > 10 and d.startswith("91"):
  26          d = d[-10:]
  27      return d if len(d) == 10 else None

  29  def to_int(v):
  30      try:
  31          return int(v)
  32      except (TypeError, ValueError):
  33          return None

  35  def parse(form):
  36      name = form.get("name", "").strip()
  37      if not name:
  38          return None, "err_student_name"
  39      phone = digits10(form.get("phone", ""))
  40      if not phone:
  41          return None, "err_phone"
  42      try:
  43          fee = int(form.get("fee", ""))
  44      except ValueError:
  45          fee = -1
  46      if fee < 0:
  47          return None, "err_fee"
  48      try:
  49          adm = date.fromisoformat(form.get("adm", "")).isoformat()
  50          end = date.fromisoformat(form.get("end", "")).isoformat()
  51      except ValueError:
  52          return None, "err_date"
  53      if end <= adm:
  54          return None, "err_end"
  55      cls = to_int(form.get("cls"))
  56      if cls not in (9, 10, 11, 12):
  57          return None, "err_student_name"
  58      ses = form.get("ses", "").strip() or cur_session()
  59      return (name, cls, form.get("parent_name", "").strip(), phone, fee, adm, ses, end), None

 119      def fetch(table, q, cls, bid):
 120          sql = (f"SELECT t.id, t.name, t.cls, t.parent_phone, t.acad_session, x.roll FROM {table} t "
 121                 "LEFT JOIN student_extra x ON x.student_id=t.id WHERE 1=1")
 122          args = []
 123          if q:
 124              like = "%" + q.replace("%", "").replace("_", "") + "%"
 125              sql += " AND (t.name LIKE ? OR x.roll LIKE ? OR t.parent_phone LIKE ?)"
 126              args += [like, like, like]
 127          if cls:
 128              sql += " AND t.cls=?"
 129              args.append(cls)
 130          if bid:
 131              sql += " AND t.id IN (SELECT student_id FROM student_batches WHERE batch_id=?)"
 132              args.append(bid)
 133          return db().execute(sql + " ORDER BY t.name COLLATE NOCASE", args).fetchall()

 136      def students():
 137          stt = request.args.get("status", "active")
 138          if stt not in ("active", "inactive", "all"):
 139              stt = "active"
 140          fee = request.args.get("fee", "")
 141          if fee not in ("paid", "pending", "upcoming"):
 142              fee = ""
 143          srt = request.args.get("sort", "name")
 144          if srt not in ("name", "recent", "due"):
 145              srt = "name"
 146          q = request.args.get("q", "").strip()[:40]
 147          cls = to_int(request.args.get("cls"))
 148          cls = cls if cls in (9, 10, 11, 12) else None
 149          bid = to_int(request.args.get("batch"))
 150          state = fee_state(db)

 152          def card(r, inactive):
 153              o, st = (0, "") if inactive else state.get(r["id"], (0, "paid"))
 154              return dict(id=r["id"], name=r["name"], phone=r["parent_phone"] or "", roll=r["roll"] or "",
 155                          ses=r["acad_session"] or "", inactive=inactive, owed=o, fst=st)
 156  
 157          rows = []
 158          if stt in ("active", "all"):
 159              rows += [card(r, False) for r in fetch("students", q, cls, bid)]
 160          if stt in ("inactive", "all"):
 161              rows += [card(r, True) for r in fetch("records", q, cls, bid)]
 162          if fee:
 163              rows = [c for c in rows if c["fst"] == fee]
 164          if srt == "recent":
 165              rows.sort(key=lambda c: c["id"], reverse=True)
 166          elif srt == "due":
 167              rows.sort(key=lambda c: (-c["owed"], c["name"].lower()))
 168          else:
 169              rows.sort(key=lambda c: c["name"].lower())
 170          bl = [dict(id=b["id"], name=b["name"]) for b in db().execute("SELECT id, name FROM batches ORDER BY name COLLATE NOCASE")]
 171          bname = next((b["name"] for b in bl if b["id"] == bid), "")
 172          nf = (stt != "active") + (fee != "") + (srt != "name")
 173  
 174          def href(**over):
 175              p = dict(status=stt, fee=fee, sort=srt, q=q, cls=cls or "", batch=bid or "")
 176              p.update(over)
 177              return "/students?" + urlencode({k: v for k, v in p.items() if v not in ("", None)})
 178  
 179          return render_template("students.html", rows=rows, status=stt, fee=fee, sort=srt, q=q,
 180                                 cls=cls, bid=bid, nf=nf, batches=bl, bname=bname, href=href)

 187          s = get_student(sid)
 188          lang = session.get("lang", "en")
 189          if lang not in MONTHS:
 190              lang = "en"

 194              "SELECT month, SUM(amount) amt FROM payments WHERE student_id=? GROUP BY month", (sid,))}

 203              up = fee

 210              (SELECT COUNT(DISTINCT y.day) FROM attendance y JOIN students t ON t.id=y.student_id
 211               WHERE t.cls=? AND y.day>=?) AS tot""", (sid, adm, s["cls"], adm)).fetchone()
